function doGet(e) { var mode = e.parameter.mode; // 'year', 'unit', 或 'random' var val = e.parameter.val; // 若是複選單元，會以逗號分隔，例如 "n數與量,a代數" 或 "普通數學" var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Questions"); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var rawData = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; var item = {}; for (var j = 0; j < headers.length; j++) { item[headers[j]] = row[j]; } rawData.push(item); } var data = []; if (mode === 'year') { data = rawData.filter(function(item) { return String(item['year']) === String(val); }); } else if (mode === 'unit') { // 支援複選，將傳入的 val 以逗號切割成陣列 var selectedUnits = val.split(',').map(function(s) { return s.trim(); }); data = rawData.filter(function(item) { var domain = String(item['domain']).trim(); var unitField = String(item['unit']).trim(); // 例如 "n數與量, a代數" // 檢查是否選了大類別全範圍 if (selectedUnits.indexOf(domain) !== -1) { return true; } // 檢查是否符合勾選的任一子項目（支援逗號分隔的多標籤） var itemUnits = unitField.split(',').map(function(s) { return s.trim(); }); for (var k = 0; k < selectedUnits.length; k++) { if (itemUnits.indexOf(selectedUnits[k]) !== -1) { return true; } } return false; }); // 嚴格去重（避免同一題因為對應多個勾選項目而被重複加入） var uniqueMap = {}; var uniqueData = []; data.forEach(function(item) { var qId = item['q_id'] || JSON.stringify(item); if (!uniqueMap[qId]) { uniqueMap[qId] = true; uniqueData.push(item); } }); data = uniqueData; } else if (mode === 'random') { data = rawData; data = shuffleArray(data); var limit = parseInt(e.parameter.limit) || 10; data = data.slice(0, limit); } return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Records"); sheet.appendRow([new Date(), data.studentName, data.studentId, data.mode, data.score, data.totalCorrect, JSON.stringify(data.details)]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()})).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function shuffleArray(array) { for (var i = array.length - 1; i > 0; i--) { var j = Math.floor(Math.random() * (i + 1)); var temp = array[i]; array[i] = array[j]; array[j] = temp; } return array; }
